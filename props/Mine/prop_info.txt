# Placed on the ground. Arms after a second, then blows up when a player steps on it.
# Whoever placed it can walk over it safely.
size: 14
armdelay: 1
lifetime: 45
damage: 35
knockback: 50

on_touch:
  blast[70]
    # The closer you are to the middle, the harder it hits
    sustain damage[victim, this.damage - distance * 0.3]
    knockback[victim]
  end
  delete prop[this]
